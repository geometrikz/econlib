\chapter{Matching}
\label{chap:matching}

Source: Gale--Shapley (1962); Roth--Sotomayor (1990), Ch.~2.

\begin{definition}[Matching, blocking pair, stability]
\label{def:stable-matching}
\lean{Econ.IsMatching, Econ.IsBlocking, Econ.IsStableMatching}
\leanok
Finite sets $M, W$ with strict preferences (injective utilities $u_m$, $u_w$); all partners acceptable. A matching is
$\mu : M \to W \cup \{\varnothing\}$ assigning no woman twice. $(m,w)$ blocks $\mu$ if $m$ strictly prefers $w$ to
$\mu(m)$ (or is unmatched) and $w$ strictly prefers $m$ to her partner (or is unmatched). $\mu$ is stable if
nothing blocks it.
\end{definition}

\begin{definition}[Deferred acceptance]
\label{def:da}
\lean{Econ.proposal, Econ.Rejected, Econ.daStep, Econ.DAInv, Econ.OptInv}
\leanok
State: for each man the set $R_m$ of women who rejected him. Each man proposes to his favourite woman outside
$R_m$; each woman keeps her favourite proposer and rejects the rest, who add her to $R_m$.
\end{definition}

\begin{lemma}[Invariants and termination]
\label{lem:da}
\lean{Econ.proposal_spec, Econ.proposal_eq_none, Econ.proposal_isSome, Econ.exists_best_proposer, Econ.subset_daStep, Econ.daInv_daStep, Econ.card_lt_daStep, Econ.sum_card_le, Econ.exists_terminal, Econ.optInv_daStep}
\leanok
\uses{def:da, def:stable-matching}
(i) Every woman in $R_m$ currently holds a proposal from a man she prefers to $m$, and this is preserved by a round.
(ii) No man is ever rejected by a woman matched to him in some stable matching (preserved by a round).
(iii) $\sum_m |R_m| \le |M||W|$ strictly increases while someone is rejected, so the algorithm stops.
\end{lemma}
\begin{proof}
\leanok
(i) A woman's favourite proposer is not rejected and keeps proposing to her, so her held proposal only improves.
(ii) If $w$ rejects $m$ for $m'$ and $\nu(m) = w$ with $\nu$ stable, then $m'$ prefers $w$ to $\nu(m')$
(all women he prefers to $w$ rejected him and are not his stable partners), and $w$ prefers $m'$ to $m$: $(m',w)$ blocks $\nu$.
\end{proof}

\begin{theorem}[Gale--Shapley]
\label{thm:gale-shapley}
\lean{Econ.exists_menOptimal_stable_matching, Econ.exists_stable_matching}
\leanok
\uses{lem:da}
A stable matching exists; the men-proposing deferred acceptance matching is stable and men-optimal: each
man weakly prefers it to his partner in any stable matching.
\end{theorem}
\begin{proof}
\leanok
At termination proposals are one-to-one (a woman with two proposers would reject one). If $m$ prefers $w$ to his
assignment then $w \in R_m$, so she holds (and is matched to) a man she prefers to $m$: no blocking pair. If
$\nu(m) = w'$ in a stable $\nu$ then $w' \notin R_m$ by (ii), so $m$'s final proposal is at least as good as $w'$.
\end{proof}

\section{Strategy-proofness of deferred acceptance}

\begin{definition}[The DA mechanism]
\label{def:da-mech}
\lean{Econ.daIter, Econ.NoRejection, Econ.daRounds, Econ.daFinal, Econ.daMatching, Econ.Improves}
\leanok
\uses{def:da}
Iterate the deferred acceptance round $|M||W| + 1$ times; the resulting matching is the DA matching for the
reported preferences.
\end{definition}

\begin{lemma}[DA trace]
\label{lem:da-trace}
\lean{Econ.daStep_eq_self, Econ.daIter_mono, Econ.daIter_inv, Econ.daIter_optInv, Econ.daIter_add_of_noRejection, Econ.exists_noRejection, Econ.daFinal_noRejection, Econ.daIter_subset_daFinal, Econ.daIter_daRounds, Econ.proposal_stay, Econ.rejected_of_mem_daIter, Econ.exists_proposer_ge, Econ.daMatching_isStable, Econ.daMatching_menOptimal}
\leanok
\uses{def:da-mech, lem:da}
Rejection sets grow, the invariants hold along the trace, a state without rejections is reached within
$|M||W| + 1$ rounds, and the DA matching is stable and men-optimal.
\end{lemma}
\begin{proof}
\leanok
As for Gale--Shapley, along an explicit sequence of states.
\end{proof}

\begin{theorem}[Blocking lemma (Gale--Sotomayor 1985) and strategy-proofness (Dubins--Freedman; Roth 1982)]
\label{thm:da-strategyproof}
\lean{Econ.gs_blocking_lemma, Econ.daMatching_strategyproof}
\leanok
\uses{lem:da-trace}
If some men $M'$ strictly prefer a matching $\mu'$ to the DA matching $\mu$, some pair $(m_0, w)$ blocks $\mu'$ with
$m_0 \notin M'$ and $w \in \mu'(M')$ (Roth--Sotomayor 1990, Lemma 3.5). Hence no man can get a strictly better partner by misreporting.
\end{theorem}
\begin{proof}
\leanok
Among the women matched under $\mu'$ to men who improve, take the one whose final proposal arrives latest; her
last rejected suitor blocks $\mu'$. A profitable misreport by $m$ would give a matching that $m$ strictly prefers
and that is stable for the reported profile, contradicting the lemma applied to the true profile.
\end{proof}

\section{Assortative matching}

\begin{theorem}[Becker 1973]
\label{thm:becker}
\lean{Econ.HasIncreasingDifferences, Econ.matchingSurplus, Econ.movedSet, Econ.becker_pam}
\leanok
With sorted types on both sides and a surplus with increasing differences on the types that occur (so, e.g.,
$\log x \log y$ on $[1,\infty)^2$ qualifies), positive assortative matching maximizes
total surplus over all one-to-one matchings.
\end{theorem}
\begin{proof}
\leanok
Let $c$ be the highest man not matched assortatively and $j$ the man matched to woman $c$; swapping the partners of
$j$ and $c$ weakly raises surplus by increasing differences and fixes one more man. Induct on the number of moved
men.
\end{proof}

\section{The housing market and top trading cycles}
\label{sec:housing-market}

\begin{definition}[TTC allocation, strict core]
\label{def:ttc}
\lean{Econ.IsTTC, Econ.WeaklyBlocks, Econ.InStrictCore}
\leanok
Each agent owns one house and has strict preferences over houses; an allocation is a permutation $\sigma$. It is a
top trading cycles allocation if there is a stage function $r$ with $r(\sigma(i)) = r(i)$ such that each $i$
receives her favourite house among owners of stage at least $r(i)$. A coalition weakly blocks $\sigma$ if it can
reallocate its own houses so that every member is weakly and some member strictly better off; the strict core is
the set of allocations no coalition weakly blocks.
\end{definition}

\begin{lemma}[Cycles]
\label{lem:exists-cycle}
\lean{Econ.exists_cycle, Econ.perm_image_eq_of_mapsTo}
\leanok
A self-map of a nonempty finite invariant set has a cycle: a nonempty invariant subset on which it is injective.
\end{lemma}
\begin{proof}
\leanok
By pigeonhole some iterate repeats, $f^a(i_0) = f^b(i_0)$ with $a < b$; the orbit of $p = f^a(i_0)$ has period
$d = b - a$, and $f^{d-1}$ inverts $f$ on it.
\end{proof}

\begin{theorem}[Roth--Postlewaite 1977; cf. Shapley--Scarf 1974]
\label{thm:ttc}
\lean{Econ.ttc_exists_on, Econ.ttc_exists, Econ.ttc_in_strictCore, Econ.strictCore_eq_ttc, Econ.ttc_unique, Econ.strictCore_nonempty}
\leanok
\uses{def:ttc, lem:exists-cycle}
With strict preferences a TTC allocation exists, it is in the strict core, and every strict-core allocation equals
it. Hence the strict core is a single allocation.
\end{theorem}
\begin{proof}
\leanok
Existence: remove a cycle of the ``point to your favourite remaining house'' map, recurse on the rest, and give the
removed cycle stage $0$. Core: by strong induction on the stage, a blocking coalition's members of stage $<k$ keep
their TTC houses; so the reallocation maps those members onto themselves, and a stage-$k$ member must receive a house
of stage $\ge k$, which she weakly prefers her TTC house to; with strict preferences nobody gains. Uniqueness: if a
core allocation $\pi$ agrees with TTC below stage $k$ but differs at stage $k$, the coalition of stages $\le k$ blocks
it with the TTC allocation.
\end{proof}

\begin{theorem}[TTC is strategy-proof and efficient; Roth 1982]
\label{thm:ttc-strategyproof}
\lean{Econ.strictCore_agree_below, Econ.ttc_strategyproof, Econ.ttc_pareto}
\leanok
\uses{thm:ttc}
With strict preferences, no agent obtains a strictly better house (by her true preferences) by reporting any other
strict ranking, and the TTC allocation is Pareto efficient.
\end{theorem}
\begin{proof}
\leanok
Agents trading before $i$'s stage keep their TTC houses in the strict core of any profile that agrees with theirs
(the induction of Theorem~\ref{thm:ttc}, using that the stage-$\le k$ coalition would otherwise block). So under a
misreport those houses are gone, and $i$ receives a house of stage at least hers, which she weakly prefers her TTC
house to. Efficiency is the core property for the grand coalition.
\end{proof}

\section{Matching with contracts}
\label{sec:contracts}

\begin{definition}[Choice functions, substitutes, IRC, stability]
\label{def:hm-stable}
\lean{Econ.ChoiceFn, Econ.ChoiceFn.R, Econ.ChoiceFn.Substitutes, Econ.ChoiceFn.IRC, Econ.IsStableAlloc, Econ.hmMap}
\leanok
Each side chooses $C(Y) \subseteq Y$ from offered contracts and rejects $R(Y) = Y \setminus C(Y)$. Substitutes: $R$ is
monotone. IRC: $C(Z) \subseteq Y \subseteq Z$ implies $C(Y) = C(Z)$. $Y$ is stable if both sides choose all of $Y$ and
no nonempty set of new contracts is chosen by both sides from $Y \cup Z$.
\end{definition}

\begin{theorem}[Hatfield--Milgrom 2005, Theorems 1 and 3]
\label{thm:hatfield-milgrom}
\lean{Econ.hm_fixedPoint_stable, Econ.hmOrderHom, Econ.stable_exists, Econ.ChoiceFn.top, Econ.ChoiceFn.all, Econ.choiceFn_top_spec}
\leanok
\uses{def:hm-stable}
If $(X_D, X_H) = (X \setminus R_H(X_H), X \setminus R_D(X_D))$ then $X_D \cap X_H$ is stable (with IRC). Under substitutes
the map is isotone in $(X_D, X_H)$ ordered by $(\subseteq, \supseteq)$, so by Tarski a fixed point, and hence a stable
allocation, exists.
\end{theorem}
\begin{proof}
\leanok
At a fixed point doctors choose $Y = X_D \cap X_H$ from $X_D$ and hospitals from $X_H$; IRC gives individual
rationality. A blocking contract $z$ lies in $X_D \setminus X_H$ or $X_H \setminus X_D$; in the first case doctors reject
it from $X_D$, hence by IRC from $Y \cup (Z \cap X_D)$ and by substitutes from $Y \cup Z$. The other case is symmetric.
\end{proof}

\begin{theorem}[Stable allocations are fixed points; side-optimal stable allocations (Hatfield--Milgrom 2005)]
\label{thm:hm-optimal}
\lean{Econ.hmFix_chooseD, Econ.hmFix_chooseH, Econ.stableXD, Econ.stableXH, Econ.choose_eq_of_rejects, Econ.stable_isFixedPoint, Econ.doctor_optimal, Econ.hospital_optimal}
\leanok
\uses{thm:hatfield-milgrom}
With substitutes and IRC on both sides, every stable $Y$ equals $X_D \cap X_H$ for the fixed point
$X_D = Y \cup \{x : x \in R_D(Y \cup \{x\}),\ x \notin R_H(Y \cup \{x\})\}$, $X_H = Y \cup \{x : x \in R_H(Y \cup \{x\})\}$.
There is a stable $Y^*$ with $C_D(Y^* \cup Y) = Y^*$ for every stable $Y$ (doctor-optimal; for unit-demand doctors with
strict preferences, every doctor weakly prefers $Y^*$), and symmetrically a hospital-optimal one.
\end{theorem}
\begin{proof}
\leanok
If $c$ chooses all of $Y \subseteq W$ and rejects each $x \in W \setminus Y$ from $Y \cup \{x\}$, then by substitutes $c(W)
\subseteq Y$ and by IRC $c(W) = Y$; so hospitals choose $Y$ from $X_H$ and doctors from $X_D$. Stability against
singletons shows that a contract outside $Y$ that doctors accept from $Y \cup \{x\}$ is rejected by hospitals; the
fixed-point equations follow by cases. For doctor-optimality take $Y^*$ from the greatest fixed point: the fixed point
of $Y$ lies below it, so $Y \cup Y^* \subseteq X_D^*$, and IRC gives $C_D(Y^* \cup Y) = C_D(X_D^*) = Y^*$. The least
fixed point is hospital-optimal in the same way.
\end{proof}
