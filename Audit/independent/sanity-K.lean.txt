/-! ### S1. Signaling: a pooling PBE that the intuitive criterion would eliminate is certified.
θL = 1, θH = 2, pH = 1/2, c(e,θ) = e/θ; pool at e = 0. Cho–Kreps eliminates every pooling
equilibrium here (any e' ∈ (1/2, 1) is equilibrium-dominated for L only), but the library has no
refinement, so nothing distinguishes this from the Riley outcome. -/
example : IsSpencePBE 1 2 (1 / 2) (fun e θ => e / θ) 0 0
    (fun x => if x = 0 then 1 / 2 else 0) := by
  refine isSpencePBE_pooling (by norm_num) (by norm_num) (fun x _ y _ hxy => ?_)
    (fun x _ y _ hxy => ?_) le_rfl (by norm_num) (by norm_num)
  · simp only; linarith
  · simp only; linarith

/-! ### S2. MoralHazard: CARA cannot be plugged in.
For u(w) = -exp(-w), φ = u⁻¹ is v ↦ -log(-v) on v < 0. No monotone φ : ℝ → ℝ agrees with it,
so `fullInsurance_optimal` / `implementsHigh_cost_gt` (which need `Monotone φ` on all of ℝ)
never apply to CARA utility. -/
example : ¬ ∃ φ : ℝ → ℝ, Monotone φ ∧ ∀ v < 0, φ v = -Real.log (-v) := by
  rintro ⟨φ, hmono, hφ⟩
  have hx0 : -Real.exp (-(φ 0 + 1)) < 0 := neg_neg_of_pos (Real.exp_pos _)
  have h1 : φ (-Real.exp (-(φ 0 + 1))) = φ 0 + 1 := by
    rw [hφ _ hx0, neg_neg, Real.log_exp]; ring
  have h2 : φ (-Real.exp (-(φ 0 + 1))) ≤ φ 0 := hmono hx0.le
  linarith
