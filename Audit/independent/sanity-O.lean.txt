import EconLean

/-!
Sanity tests for the statement review of Behavioral/* and BoundedRationality/BerkNash.lean.
Each `example` either (a) confirms a hypothesis set is satisfiable on a nondegenerate object,
(b) shows a hypothesis is genuinely needed / a result is silent outside it, or
(c) states a canonical result that is NOT in the targets (coverage gap). Hard proofs are `sorry`.
-/

open Econ

/-! ## 1. Rational inattention: uniform marginals are ASSUMED (`hunif`), not derived. -/

section RI
open Econ.RationalInattention Econ.Logit

-- (a) One state, two actions, v = (1,0), λ = 1. Mutual information is 0 for every rule, so the
-- optimum is the point mass on action 0; its marginal is (1,0), not uniform, so
-- `optimal_is_softmax_of_uniform` says nothing, and indeed softmax is NOT optimal here.
example :
    IsOptimalRule 1 (fun _ : Unit => (1 : ℝ)) (fun _ => ![1, 0]) (fun _ => ![1, 0]) ∧
      (fun _ : Unit => (![1, 0] : Fin 2 → ℝ)) ≠ fun _ => softmax 1 ![1, 0] := by
  sorry

-- (b) The canonical Matějka–McKay (2015) corollary that the targets do NOT contain:
-- a priori homogeneous options (prior invariant under permutations of actions) ⇒ an optimal
-- rule is the plain multinomial logit. Here invariance is encoded as: every action permutation
-- is matched by a prior-preserving relabelling of states.
example {Ω ι : Type*} [Fintype Ω] [Fintype ι] [Nonempty ι] [DecidableEq Ω] {lam : ℝ}
    (hlam : 0 < lam) {g : Ω → ℝ} (hg : IsPrior g) (v : Ω → ι → ℝ)
    (hsym : ∀ σ : Equiv.Perm ι, ∃ τ : Equiv.Perm Ω,
      (∀ ω, g (τ ω) = g ω) ∧ ∀ ω i, v (τ ω) (σ i) = v ω i) :
    ∃ p, IsOptimalRule lam g v p ∧ ∀ ω, p ω = softmax lam (v ω) := by
  sorry

-- (c) `hunif` is at least satisfiable (v ≡ 0, the uniform state-independent rule is optimal).
example {Ω ι : Type*} [Fintype Ω] [Fintype ι] [Nonempty ι] {g : Ω → ℝ} (hg : IsPrior g) :
    IsOptimalRule 1 g (fun _ _ => 0) (fun _ _ => 1 / (Fintype.card ι : ℝ)) ∧
      ∀ i, marginal g (fun _ _ => 1 / (Fintype.card ι : ℝ)) i = 1 / (Fintype.card ι : ℝ) := by
  sorry

end RI

/-! ## 2. Kőszegi–Rabin: `exists_ppe` still takes PE existence as a hypothesis. -/

section KR
open Econ.ReferenceDependence

-- For an unrestricted gain–loss function μ and a finite choice set, a PE can fail to exist
-- (3-cycle of profitable deviations 0 → 1 → 2 → 0), so `hPE` in `exists_ppe` is a real
-- assumption, not a derivable one. Existence is only derived in `exists_pe` /
-- `exists_ppe_of_convex` (convex compact D, i.e. randomisation allowed) and in
-- `exists_ppeDet` (deterministic, piecewise-linear μ with λ ≥ 1).
example :
    let μ : ℝ → ℝ := fun x => if x = -2 then 3 else 0
    let D : Set (Unit → ℝ) := {fun _ => 0, fun _ => 1, fun _ => 2}
    ∀ c, ¬ IsPEDet μ (fun _ x => x) D c := by
  sorry

-- The same cycle with degenerate lotteries (via `U_pure`) gives a finite lottery set D with no PE.
example :
    let μ : ℝ → ℝ := fun x => if x = -2 then 3 else 0
    let bundle : Fin 3 → Unit → ℝ := fun i _ => (i : ℝ)
    let D : Set (Fin 3 → ℝ) := {Pi.single 0 1, Pi.single 1 1, Pi.single 2 1}
    ∀ F, ¬ IsPE bundle μ (fun _ x => x) D F := by
  sorry

-- Nondegenerate check of the derived existence result (no PE hypothesis needed).
example : ∃ F, IsPPE (fun (i : Fin 2) (_ : Fin 1) => (i : ℝ)) (μLin 1 2) (fun _ x => x)
    (allLotteries (Fin 2)) F :=
  exists_ppe_allLotteries _ _ _

end KR

/-! ## 3. Falmagne / Luce: coverage check (Luce ⇒ RUM is not stated). -/

section SC
open Econ.StochasticChoice

-- Block–Marschak / Holman–Marley: every Luce rule with positive weights is a RUM.
-- Derivable from `isRUM_iff_bm_nonneg` but not stated in the targets.
example {X : Type*} [Fintype X] [DecidableEq X] {w : X → ℝ} (hw : ∀ a, 0 < w a) :
    ∃ μ : Ranking X → ℝ, IsRUM μ ∧ ∀ A : Finset X, A.Nonempty → ∀ a, rumP μ A a = luce w A a := by
  sorry

-- Trivial instantiation: one alternative. Fine (theorem is not vacuous; just a sanity probe).
example (P : Finset Unit → Unit → ℝ) (hP : IsChoiceRule P) :
    ∃ μ : Ranking Unit → ℝ, IsRUM μ ∧ ∀ A : Finset Unit, A.Nonempty → ∀ a, rumP μ A a = P A a :=
  falmagne hP (by sorry)

end SC

/-! ## 8. Gabaix cognitive discounting: forward-guidance result is IS-curve only. -/

section CD
open Econ.CognitiveNK.Model

-- `fg_response` uses `IsISPath` (IS equation alone, inflation/Phillips curve absent): the
-- rational benchmark gives x 0 = 1/σ for every horizon T.

  exact ⟨x, hx, he, fg_puzzle_rational hx he⟩
