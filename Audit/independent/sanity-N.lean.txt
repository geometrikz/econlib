/-- BK coverage gap: a saddle-path system (one predetermined, one jump variable) is
BK-determinate, yet `DeterminateN` fails, because it treats every variable as a jump variable. -/
example : ¬ NK.DeterminateN (Matrix.diagonal ![(1 / 2 : ℝ), 2]) := by
  intro h
  have hz := h (fun t => ![(1 / 2 : ℝ) ^ t, 0])
    (fun t => by
      funext i; rw [mulVec_diagonal]
      fin_cases i <;> simp [pow_succ] <;> ring)
    ⟨1, fun t i => by
      fin_cases i
      · simp only [Fin.zero_eta, Matrix.cons_val_zero]
        rw [abs_of_pos (by positivity)]; exact pow_le_one₀ (by norm_num) (by norm_num)
      · simp⟩ 0
  simpa using congrFun hz 0

/-- SCC MISMATCH: the Lean SCC differs from GHKT's γ·Y₀·kernel whenever s > 0. -/
theorem scc_ne_ghkt (P : Climate.Params) (hs : 0 < P.s) (hL : 0 < P.φL)
    {Y S : ℕ → ℝ} (hY : ∀ j, 0 < Y j) :
    P.scc Y S ≠ P.γ * P.outputAfter Y S 0 0 * P.kernel := by
  rw [P.scc_eq hY]
  intro h
  have hk := P.kernel_pos_of hL
  have hY0 : 0 < P.outputAfter Y S 0 0 := by
    unfold Climate.Params.outputAfter; exact mul_pos (Real.exp_pos _) (hY 0)
  have hγ := P.hγ
  have : 0 < P.γ * P.outputAfter Y S 0 0 * P.kernel * P.s := by positivity
  nlinarith

/-- SLP Thm 11.4 is not covered: Q = [[1,0],[1,0]] has a strictly positive column
(and a unique invariant distribution), but every power has a zero entry, so the
hypothesis of `doeblin_pow` (all entries of Q^N ≥ ε > 0) never holds. -/
theorem slp114_not_covered :
    ∀ N : ℕ, (!![(1 : ℝ), 0; 1, 0] ^ N) 0 1 = 0 := by
  intro N
  induction N with
  | zero => simp
  | succ N ih =>
    rw [pow_succ, Matrix.mul_apply, Fin.sum_univ_two]
    simp

example (N : ℕ) : ¬ ∃ ε : ℝ, 0 < ε ∧ ∀ z z', ε ≤ (!![(1 : ℝ), 0; 1, 0] ^ N) z z' := by
  rintro ⟨ε, hε, h⟩
  have := h 0 1
  rw [slp114_not_covered N] at this
  linarith
